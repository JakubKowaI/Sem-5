\section*{Gramatyka}
Rozważana gramatyka (z rozszerzonym symbolem startowym $E'$):
\begin{center}
\begin{enumerate}
\item E' $\to$ E \\
\item E $\to$ E\ \text{or}\ T \\
\item E $\to$ T \\
\item T $\to$ T\ \text{and}\ F \\
\item T $\to$ F \\
\item F $\to$ \text{not}\ F \\
\item F $\to$ (E) \\
\item F $\to$ \text{true} \\
\item F $\to$ \text{false}
\end{enumerate}
\end{center}

\section*{Kolekcja kanoniczna zbiorów LR(0)}
$I_0 = closure({E' \to \cdot E})$
\begin{align*}
E' &\to \cdot E \\
E &\to \cdot E\ \text{or}\ T \\
E &\to \cdot T \\
T &\to \cdot T\ \text{and}\ F \\
T &\to \cdot F \\
F &\to \cdot \text{not}\ F \\
F &\to \cdot (E) \\
F &\to \cdot \text{true} \\
F &\to \cdot \text{false}
\end{align*}

\subsection*{Przejścia (GOTO)}
Zdefiniowane przejścia pomiędzy stanami (skrócona lista):
